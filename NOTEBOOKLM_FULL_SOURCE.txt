#!/bin/bash
# ==============================================================================
# [1차]: 공식 바이너리(agent-leak-app-x86) 실행을 위한 WSL2 Ubuntu 사전 요구사항 검증 및 환경 구축 스크립트입니다.

set -euo pipefail

if [ "$(id -u)" -eq 0 ]; then
    echo "[ERROR] root 계정으로 실행할 수 없습니다. 일반 사용자(agent-admin)로 실행하세요."
    exit 1

AGENT_HOME="${AGENT_HOME:-/home/agent-admin/agent-app}"
AGENT_PORT="${AGENT_PORT:-15034}"
AGENT_UPLOAD_DIR="${AGENT_UPLOAD_DIR:-${AGENT_HOME}/upload_files}"

BIN_NAME="agent-leak-app-x86"
BIN_PATH="${AGENT_HOME}/${BIN_NAME}"

mkdir -p "${AGENT_HOME}/bin" "${AGENT_HOME}/scripts" "${AGENT_UPLOAD_DIR}" "${AGENT_KEY_PATH}" "${AGENT_LOG_DIR}"

    echo "[ERROR] 실행 가능한 공식 바이너리가 없습니다: ${BIN_PATH}"
    exit 1

pkill -9 -f "agent-leak-app" 2>/dev/null || true
pkill -9 -f "agent_app.py" 2>/dev/null || true

if ss -tlpn 2>/dev/null | grep -q ":${AGENT_PORT} "; then
    echo "[WARN] 포트 ${AGENT_PORT} 점유 프로세스 정리 중..."
    fuser -k "${AGENT_PORT}/tcp" 2>/dev/null || true
    sleep 0.5

echo "[PASS] 바이너리 경로: ${BIN_PATH}"

#!/bin/bash
# ==============================================================================
# [1차]: agent-leak-app-x86 프로세스의 생존 여부, CPU/메모리 사용량, 로그 정체 시간(LOG_IDLE)을 감시하는 관제 스크립트입니다.
# [2차]: 서버라는 병실에 들어간 환자의 심박수(PID), 체온(CPU), 혈압(RSS 메모리), 숨쉬는 간격(LOG_IDLE)을 체크해 진료 차트(monitor.log)에 적는 간호사입니다.
# ==============================================================================

set -u

TARGET_PATTERN="${TARGET_PATTERN:-agent-leak-app-x86}"
LOG_DIR="${AGENT_LOG_DIR:-/var/log/agent-app}"
MONITOR_LOG="${MONITOR_LOG:-${LOG_DIR}/monitor.log}"
APP_LOG="${APP_LOG:-${LOG_DIR}/agent_app.log}"

if [ ! -f "${APP_LOG}" ] && [ -f "${LOG_DIR}/app.log" ]; then
    APP_LOG="${LOG_DIR}/app.log"
fi

mkdir -p "${LOG_DIR}"
NOW_STR="$(date '+%Y-%m-%d %H:%M:%S')"
PID="$(pgrep -n -f "${TARGET_PATTERN}" || true)"

if [ -z "${PID}" ]; then
    echo "[${NOW_STR}] PROCESS:down TARGET:${TARGET_PATTERN}" >> "${MONITOR_LOG}"
    exit 1
fi

STAT="$(ps -o stat= -p "${PID}" 2>/dev/null | awk '{print $1}')"
CPU_PCT="$(ps -o pcpu= -p "${PID}" 2>/dev/null | awk '{print $1}')"
MEM_PCT="$(ps -o pmem= -p "${PID}" 2>/dev/null | awk '{print $1}')"
RSS_MB="$(ps -o rss= -p "${PID}" 2>/dev/null | awk '{print int($1/1024)}')"

if [ -z "${STAT}" ]; then
    echo "[${NOW_STR}] PROCESS:down TARGET:${TARGET_PATTERN}" >> "${MONITOR_LOG}"
    exit 1
fi

LOG_IDLE=0
if [ -f "${APP_LOG}" ]; then
    NOW_EPOCH="$(date +%s)"
    MTIME_EPOCH="$(stat -c %Y "${APP_LOG}" 2>/dev/null || echo "${NOW_EPOCH}")"
    LOG_IDLE=$(( NOW_EPOCH - MTIME_EPOCH ))
fi

echo "[${NOW_STR}] PID:${PID} STAT:${STAT} CPU:${CPU_PCT}% RSS:${RSS_MB}MB MEM:${MEM_PCT}% LOG_IDLE:${LOG_IDLE}s" >> "${MONITOR_LOG}"

#!/bin/bash
# ==============================================================================
# [1차]: 공식 바이너리(agent-leak-app-x86)를 WSL2에서 실제로 실행해 OOM, CPU, Deadlock Before/After 로그를 수집합니다.
# [2차]: 3가지 고장 상황을 실험실에서 직접 재현하고, 관제 기록(monitor.log)과 진료 차트(agent_app.log)를 파일로 봉인하는 자동 실험기입니다.
# ==============================================================================

set -euo pipefail

REPO_ROOT="$(pwd)"
LOG_ARCHIVE_DIR="${REPO_ROOT}/logs"
mkdir -p "${LOG_ARCHIVE_DIR}"

export AGENT_HOME="/home/agent-admin/agent-app"
export AGENT_PORT="15034"
export AGENT_UPLOAD_DIR="${AGENT_HOME}/upload_files"
export AGENT_KEY_PATH="${AGENT_HOME}/api_keys"
export AGENT_LOG_DIR="/var/log/agent-app"
BIN_PATH="${AGENT_HOME}/agent-leak-app-x86"
MONITOR_SCRIPT="${AGENT_HOME}/scripts/monitor.sh"

mkdir -p "${AGENT_HOME}/scripts"
sed 's/\r$//' "${REPO_ROOT}/scripts/monitor.sh" > "${MONITOR_SCRIPT}"
chmod +x "${MONITOR_SCRIPT}"

cleanup_all() {
    pkill -9 -f "agent-leak-app" 2>/dev/null || true
    pkill -9 -f "agent_app.py" 2>/dev/null || true
    pkill -9 -f "monitor.sh" 2>/dev/null || true
    fuser -k "${AGENT_PORT}/tcp" 2>/dev/null || true
    sleep 1
}

run_case() {
    local case_name="$1"
    local mem_limit="$2"
    local cpu_limit="$3"
    local multi_thread="$4"
    local max_wait="$5"

    echo "=== Running ${case_name} (MEM=${mem_limit}, CPU=${cpu_limit}, THREAD=${multi_thread}) ==="
    cleanup_all
    : > "${AGENT_LOG_DIR}/agent_app.log"
    : > "${AGENT_LOG_DIR}/monitor.log"

    export MEMORY_LIMIT="${mem_limit}"
    export CPU_MAX_OCCUPY="${cpu_limit}"
    export MULTI_THREAD_ENABLE="${multi_thread}"

    local start_ts
    start_ts="$(date +%s)"

    "${BIN_PATH}" > "${LOG_ARCHIVE_DIR}/${case_name}_stdout.log" 2>&1 &
    local app_pid=$!

    (
        while kill -0 "${app_pid}" 2>/dev/null; do
            "${MONITOR_SCRIPT}" || true
            sleep 1
        done
        "${MONITOR_SCRIPT}" || true
    ) &
    local mon_pid=$!

    local elapsed=0
    local exit_code=0
    while kill -0 "${app_pid}" 2>/dev/null; do
        if [ "${elapsed}" -ge "${max_wait}" ]; then
            if [ "${case_name}" = "deadlock_before" ]; then
                ps -ef | grep "agent-leak-app-x86" | grep -v grep > "${LOG_ARCHIVE_DIR}/${case_name}_ps_ef.txt" || true
                ps -eLf | grep "${app_pid}" | grep -v grep > "${LOG_ARCHIVE_DIR}/${case_name}_ps_elf.txt" || true
                top -b -n 1 -H -p "${app_pid}" > "${LOG_ARCHIVE_DIR}/${case_name}_top_h.txt" || true
            fi
            kill -9 "${app_pid}" 2>/dev/null || true
            exit_code=124
            break
        fi
        if [ "${case_name}" = "cpu_before" ] && [ "${elapsed}" -eq 3 ]; then
            top -b -n 1 -p "${app_pid}" > "${LOG_ARCHIVE_DIR}/${case_name}_top.txt" || true
        fi
        sleep 1
        elapsed=$((elapsed + 1))
    done

    if [ "${exit_code}" -eq 0 ]; then
        wait "${app_pid}" || exit_code=$?
    fi
    wait "${mon_pid}" 2>/dev/null || true

    local end_ts
    end_ts="$(date +%s)"
    local duration=$((end_ts - start_ts))

    cp "${AGENT_LOG_DIR}/agent_app.log" "${LOG_ARCHIVE_DIR}/${case_name}_app.log"
    cp "${AGENT_LOG_DIR}/monitor.log" "${LOG_ARCHIVE_DIR}/${case_name}_monitor.log"
    echo "${exit_code},${duration}" > "${LOG_ARCHIVE_DIR}/${case_name}_meta.txt"
    echo "[DONE] ${case_name}: exit=${exit_code}, duration=${duration}s"
}

# 1. OOM: Before (128MB) vs After (512MB)
run_case "oom_before" 128 30 false 60
run_case "oom_after" 512 30 false 35

# 2. CPU: Before (80%) vs After (30%)
run_case "cpu_before" 512 80 false 60
run_case "cpu_after" 512 30 false 35

# 3. Deadlock: Before (true) vs After (false)
run_case "deadlock_before" 512 30 true 12
run_case "deadlock_after" 512 30 false 12

cleanup_all
echo "ALL REAL BINARY EXPERIMENTS COMPLETED."

#!/usr/bin/env python3
"""실제 agent-leak-app-x86 실행 로그(agent_app.log, monitor.log)를 파싱해 JSON 요약 결과를 생성하는 모듈."""

from __future__ import annotations

import json
import re
from pathlib import Path
from typing import Any


RSS_PATTERN = re.compile(r"RSS:(\d+)MB")
CPU_PATTERN = re.compile(r"CPU:([0-9.]+)%")
HEAP_PATTERN = re.compile(r"Current Heap:\s*(\d+)MB")
LOAD_PATTERN = re.compile(r"Current Load:\s*([0-9.]+)%")
IDLE_PATTERN = re.compile(r"LOG_IDLE:(\d+)s")


def parse_monitor_log(text: str) -> dict[str, Any]:
    """monitor.log 텍스트에서 최대 RSS, 최대 CPU, 최대 LOG_IDLE, 다운 감지 여부를 추출합니다."""
    # [1차]: 정규표현식 finditer로 로그 전체에서 RSS, CPU, LOG_IDLE 수치를 리스트로 추출합니다.
    # [2차]: 간호사가 적어둔 진료 차트에서 혈압(RSS), 체온(CPU), 호흡 멈춤 시간(LOG_IDLE) 숫자만 쏙쏙 뽑아냅니다.
    rss_values = [int(m.group(1)) for m in RSS_PATTERN.finditer(text)]
    cpu_values = [float(m.group(1)) for m in CPU_PATTERN.finditer(text)]
    idle_values = [int(m.group(1)) for m in IDLE_PATTERN.finditer(text)]
    # [1차]: 프로세스 종료 마커("PROCESS:down") 문자열 포함 여부를 불리언으로 판별합니다.
    # [2차]: 차트 맨 아래에 "환자 퇴실(PROCESS:down)" 도장이 찍혔는지 확인합니다.
    process_down = "PROCESS:down" in text

    return {
        "samples": len(rss_values),
        "peak_rss_mb": max(rss_values) if rss_values else 0,
        "min_rss_mb": min(rss_values) if rss_values else 0,
        "peak_cpu_pct": max(cpu_values) if cpu_values else 0.0,
        "max_log_idle_sec": max(idle_values) if idle_values else 0,
        "process_down_detected": process_down,
    }


def parse_app_log(text: str) -> dict[str, Any]:
    """agent_app.log 텍스트에서 핵심 이벤트(OOM, Cache Flushed, Watchdog, Deadlock)를 추출합니다."""
    heap_values = [int(m.group(1)) for m in HEAP_PATTERN.finditer(text)]
    load_values = [float(m.group(1)) for m in LOAD_PATTERN.finditer(text)]

    return {
        "line_count": len(text.splitlines()),
        "peak_heap_mb": max(heap_values) if heap_values else 0,
        "peak_internal_load_pct": max(load_values) if load_values else 0.0,
        "oom_self_terminated": "SELF-TERMINATED" in text or "Self-terminating" in text,
        "memory_cache_flushed": "Memory Cache Flushed" in text or "Flushed" in text,
        "watchdog_sigterm": ("WATCHDOG" in text and "SIGTERM" in text) or "CPU Threshold Violated" in text,
        "cooldown_detected": "Cooldown" in text or "cooldown" in text.lower(),
        "deadlock_detected": (
            "Deadlock" in text
            or "BLOCKED" in text
            or "waiting" in text.lower()
            or "Contention" in text
        ),
        "healthy_completed": "All tasks completed" in text,
    }


def summarize_experiment_pair(
    experiment_name: str,
    before_app_log: str,
    before_monitor_log: str,
    before_exit_code: int,
    before_duration_sec: float,
    after_app_log: str,
    after_monitor_log: str,
    after_exit_code: int,
    after_duration_sec: float,
) -> dict[str, Any]:
    """Before/After 실험 로그와 종료 코드를 종합하여 JSON 구조체로 반환합니다."""
    return {
        "experiment": experiment_name,
        "target_binary": "agent-leak-app-x86",
        "before": {
            "exit_code": before_exit_code,
            "duration_sec": round(before_duration_sec, 2),
            "app": parse_app_log(before_app_log),
            "monitor": parse_monitor_log(before_monitor_log),
        },
        "after": {
            "exit_code": after_exit_code,
            "duration_sec": round(after_duration_sec, 2),
            "app": parse_app_log(after_app_log),
            "monitor": parse_monitor_log(after_monitor_log),
        },
    }


def save_json_summary(output_path: Path, payload: dict[str, Any]) -> None:
    """요약 딕셔너리를 UTF-8 JSON 파일로 저장합니다."""
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(json.dumps(payload, indent=2, ensure_ascii=False), encoding="utf-8")


def _read_meta(meta_path: Path) -> tuple[int, float]:
    """<case>_meta.txt에서 (exit_code, duration_sec)를 읽어옵니다."""
    if not meta_path.exists():
        return -1, 0.0
    parts = meta_path.read_text(encoding="utf-8").strip().split(",")
    return int(parts[0]), float(parts[1])


def generate_all_results(repo_root: Path | None = None) -> list[Path]:
    """logs/ 폴더의 실측 로그를 파싱해 experiment_results/*.json 3종을 생성합니다."""
    root = repo_root or Path(__file__).resolve().parent.parent
    logs_dir = root / "logs"
    out_dir = root / "experiment_results"
    cases = [
        ("oom", "oom_before", "oom_after", "oom_experiment_result.json"),
        ("cpu", "cpu_before", "cpu_after", "cpu_experiment_result.json"),
        ("deadlock", "deadlock_before", "deadlock_after", "deadlock_experiment_result.json"),
    ]
    written: list[Path] = []
    for name, before_key, after_key, out_name in cases:
        b_app = (logs_dir / f"{before_key}_app.log").read_text(encoding="utf-8", errors="replace")
        b_mon = (logs_dir / f"{before_key}_monitor.log").read_text(encoding="utf-8", errors="replace")
        b_exit, b_dur = _read_meta(logs_dir / f"{before_key}_meta.txt")

        a_app = (logs_dir / f"{after_key}_app.log").read_text(encoding="utf-8", errors="replace")
        a_mon = (logs_dir / f"{after_key}_monitor.log").read_text(encoding="utf-8", errors="replace")
        a_exit, a_dur = _read_meta(logs_dir / f"{after_key}_meta.txt")

        summary = summarize_experiment_pair(name, b_app, b_mon, b_exit, b_dur, a_app, a_mon, a_exit, a_dur)
        out_path = out_dir / out_name
        save_json_summary(out_path, summary)
        written.append(out_path)
    return written


if __name__ == "__main__":
    for p in generate_all_results():
        print(f"Saved: {p}")

def _draw_box(
    ax: plt.Axes,
    x: float,
    y: float,
    w: float,
    h: float,
    title: str,
    body: str,
    fc: str,
    ec: str,
    title_color: str = "#1A202C",
) -> None:
    box = FancyBboxPatch(
        (x, y),
        w,
        h,
        boxstyle="round,pad=0.3,rounding_size=0.8",
        facecolor=fc,
        edgecolor=ec,
        linewidth=1.8,
    )
    ax.add_patch(box)
    ax.text(
        x + w / 2,
        y + h - 2.3,
        title,
        ha="center",
        va="top",
        fontsize=10.5,
        fontweight="bold",
        color=title_color,
    )
    ax.text(
        x + w / 2,
        y + h / 2 - 1.0,
        body,
        ha="center",
        va="center",
        fontsize=8.8,
        color="#2D3748",
    )


def _arrow(ax: plt.Axes, x1: float, y1: float, x2: float, y2: float, label: str = "", color: str = "#4A5568") -> None:
    ax.annotate(
        "",
        xy=(x2, y2),
        xytext=(x1, y1),
        arrowprops={"arrowstyle": "->", "lw": 1.8, "color": color},
    )
    if label:
        ax.text(
            (x1 + x2) / 2,
            (y1 + y2) / 2 + 1.2,
            label,
            ha="center",
            va="bottom",
            fontsize=8.2,
            fontweight="bold",
            color=color,
        )

#!/usr/bin/env python3
"""평가자용 원클릭 요구사항 & 실측 로그 통합 검증 러너."""

from __future__ import annotations

import json
import subprocess
import sys
from pathlib import Path

if hasattr(sys.stdout, "reconfigure"):
    sys.stdout.reconfigure(encoding="utf-8")

REPO_ROOT = Path(__file__).resolve().parent.parent


def check_file(rel_path: str) -> tuple[bool, str]:
    p = REPO_ROOT / rel_path
    if p.exists() and p.is_file() and p.stat().st_size > 0:
        return True, f"{p.stat().st_size:,} bytes"
    return False, "Missing or empty"


def check_json(rel_path: str) -> tuple[bool, str]:
    p = REPO_ROOT / rel_path
    if not p.exists():
        return False, "Missing JSON"
    data = json.loads(p.read_text(encoding="utf-8"))
    b_exit = data["before"]["exit_code"]
    a_exit = data["after"]["exit_code"]
    return True, f"Before exit={b_exit} -> After exit={a_exit}"


def run_unit_tests() -> tuple[bool, str]:
    proc = subprocess.run(
        [sys.executable, "-m", "unittest", "discover", "-s", "tests"],
        cwd=REPO_ROOT,
        capture_output=True,
        text=True,
    )
    if proc.returncode == 0:
        return True, "All unit tests passed"
    return False, proc.stderr.strip() or "Unit tests failed"


def main() -> int:
    checks: list[tuple[str, str, tuple[bool, str]]] = []

    required_files = [
        ("Docs", "PLAN.md"),
        ("Docs", "스터디.md"),
        ("Docs", "REPORT.md"),
        ("Docs", "NOTEBOOKLM_GUIDE.md"),
        ("Docs", "NOTEBOOKLM_FULL_SOURCE.txt"),
        ("Issues", "issues/01_oom_crash_issue.md"),
        ("Issues", "issues/02_cpu_latency_issue.md"),
        ("Issues", "issues/03_deadlock_issue.md"),
        ("Scripts", "scripts/setup_environment.sh"),
        ("Scripts", "scripts/monitor.sh"),
        ("Scripts", "scripts/run_experiments.sh"),
        ("Scripts", "scripts/log_analyzer.py"),
        ("Scripts", "scripts/generate_diagrams_matplotlib.py"),
        ("Scripts", "scripts/build_notebooklm_source.py"),
        ("Diagrams", "diagrams/01_system_architecture_overview.png"),
        ("Diagrams", "diagrams/02_oom_memory_leak_mechanism.png"),
        ("Diagrams", "diagrams/03_cpu_watchdog_mechanism.png"),
        ("Diagrams", "diagrams/04_deadlock_circular_wait.png"),
        ("Real Logs", "logs/oom_before_app.log"),
        ("Real Logs", "logs/cpu_before_app.log"),
        ("Real Logs", "logs/deadlock_before_app.log"),
    ]

    for cat, rel in required_files:
        checks.append((cat, rel, check_file(rel)))

    for rel in [
        "experiment_results/oom_experiment_result.json",
        "experiment_results/cpu_experiment_result.json",
        "experiment_results/deadlock_experiment_result.json",
    ]:
        checks.append(("Results", rel, check_json(rel)))

    checks.append(("Tests", "tests/test_log_analyzer.py", run_unit_tests()))

    print("=" * 78)
    print("  [EVALUATOR RUNNER] Codyssey B4-02 Official Binary Verification")
    print("=" * 78)
    print(f"{'Category':<10} | {'Item':<44} | {'Status':<6} | Details")
    print("-" * 78)

    passed = 0
    for cat, item, (ok, detail) in checks:
        status = "PASS" if ok else "FAIL"
        if ok:
            passed += 1
        print(f"{cat:<10} | {item:<44} | [{status}] | {detail}")

    total = len(checks)
    print("=" * 78)
    print(f"  Total Score: {passed}/{total} Passed ({(passed / total) * 100:.1f}%)")
    print("=" * 78)
    return 0 if passed == total else 1
